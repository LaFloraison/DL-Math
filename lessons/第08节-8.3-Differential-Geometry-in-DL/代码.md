# §8.3 微分几何在深度学习中的应用 · 代码

主线里我们看见：微分几何给深度学习提供了四片透镜——流形（数据在哪）、切空间（往哪走）、度量（走多远）、曲率（走到的位置稳不稳）。这节代码，把这些透镜装进一个**亲手训练的微型神经网络**上，走一遍完整的"几何体检"：① 造一个两类的 2D 数据；② 用手写反向传播训练一个 2-4-1 的小网络；③ **透镜一（流形/特征空间）**：看隐藏层特征在 PCA 投影下如何聚团分离；④ **透镜二/三（切空间 + 损失景观）**：在权重空间切一 2D 切片，画损失地形，看它是什么形状；⑤ **透镜四（曲率）**：在收敛点算 Hessian 特征值，判断落到的是平坦还是尖锐极小值。

需要的环境：Python 3.8+ 与 `numpy`、`matplotlib`。

## 层一：完整代码块

复制粘贴即可运行。输出直接印在代码下方（`SEED = 0` 固定，结果可复现）。

```python
"""
§8.3 微分几何在深度学习中的应用 · 代码.md
训练一个微型 MLP(2-4-1)，做"几何体检"：
特征空间聚类 / 损失景观切片 / 收敛点曲率(Hessian 特征值)。

运行：python 代码.py    （依赖 matplotlib）
"""

import numpy as np
import matplotlib.pyplot as plt

SEED = 0
N_CLASS = 60            # 每类样本数
H = 4                   # 隐藏层神经元数
STEPS, ETA = 2000, 0.3

rng = np.random.default_rng(SEED)


def make_two_clusters(n=N_CLASS, seed=SEED):
    """两类数据：A 在 (1.5,1.5) 附近，B 在 (-1.5,-1.5) 附近。"""
    rng = np.random.default_rng(seed)
    A = rng.standard_normal((n, 2)) * 0.8 + np.array([1.5, 1.5])
    B = rng.standard_normal((n, 2)) * 0.8 + np.array([-1.5, -1.5])
    X = np.vstack([A, B])                     # (2n, 2) 特征
    y = np.concatenate([np.ones(n), np.zeros(n)])  # (2n,) 标签 {0,1}
    return X, y


def init_weights(seed=SEED):
    """初始化 MLP 权重：W1(4,2), b1(4,), W2(1,4), b2(1,)。"""
    rng = np.random.default_rng(seed)
    W1 = rng.standard_normal((H, 2)) * 0.5
    b1 = rng.standard_normal(H) * 0.5
    W2 = rng.standard_normal((1, H)) * 0.5
    b2 = rng.standard_normal(1) * 0.5
    return W1, b1, W2, b2


def forward(X, W1, b1, W2, b2):
    """前向：h = ReLU(W1 X + b1);  yhat = sigmoid(W2 h + b2)。"""
    z1 = X @ W1.T + b1                         # (2n, H)
    h = np.maximum(z1, 0.0)                    # ReLU
    z2 = h @ W2.T + b2                         # (2n, 1)
    yhat = 1.0 / (1.0 + np.exp(-z2))           # sigmoid
    return z1, h, z2, yhat


def loss_and_grad(X, y, W1, b1, W2, b2):
    """二分类交叉熵损失 + 手写反向传播。返回 (loss, 各权重梯度)。"""
    n = X.shape[0]
    z1, h, z2, yhat = forward(X, W1, b1, W2, b2)
    loss = -np.mean(y * np.log(yhat + 1e-12) + (1 - y) * np.log(1 - yhat + 1e-12))
    dz2 = (yhat - y.reshape(-1, 1)) / n        # (2n, 1)
    dW2 = dz2.T @ h                            # (1, H)
    db2 = dz2.sum(axis=0, keepdims=True)       # (1,)
    dz1 = (dz2 @ W2) * (z1 > 0)                # 通过 ReLU 掩码
    dW1 = dz1.T @ X                            # (H, 2)
    db1 = dz1.sum(axis=0)                      # (H,)
    return loss, (dW1, db1, dW2, db2)


def flatten(params):
    """把 (W1,b1,W2,b2) 拍平成权重向量。"""
    W1, b1, W2, b2 = params
    return np.concatenate([W1.ravel(), b1, W2.ravel(), b2])


def unflatten(vec):
    """权重向量还原为 (W1,b1,W2,b2)。"""
    i = 0
    W1 = vec[i:i+H*2].reshape(H, 2); i += H*2
    b1 = vec[i:i+H]; i += H
    W2 = vec[i:i+H].reshape(1, H); i += H
    b2 = vec[i:i+1]
    return W1, b1, W2, b2


def predict(X, W1, b1, W2, b2):
    _, h, _, yhat = forward(X, W1, b1, W2, b2)
    return h, yhat


def hidden_features(X, params):
    """透镜一：隐藏层激活（特征空间）。"""
    W1, b1, W2, b2 = unflatten(params)
    _, h, _, _ = forward(X, W1, b1, W2, b2)
    return h


def loss_slice(params0, X, y, d1, d2, grid=40):
    """透镜二/三：沿两个权重方向 d1,d2 切损失景观。"""
    xs = np.linspace(-1.5, 1.5, grid)
    L = np.zeros((grid, grid))
    for i, a in enumerate(xs):
        for j, b in enumerate(xs):
            p = unflatten(params0 + a * d1 + b * d2)
            L[i, j], _ = loss_and_grad(X, y, *p)
    return xs, L


def hessian_at(params, X, y, eps=1e-5):
    """透镜四：用中心差分算损失在 params 处的 Hessian 特征值。"""
    n_params = params.size
    H = np.zeros((n_params, n_params))
    for i in range(n_params):
        for j in range(n_params):
            if j < i:
                H[i, j] = H[j, i]; continue
            ei, ej = np.zeros(n_params), np.zeros(n_params)
            ei[i], ej[j] = eps, eps
            Lpp, _ = loss_and_grad(X, y, *unflatten(params + ei + ej))
            Lpm, _ = loss_and_grad(X, y, *unflatten(params + ei - ej))
            Lmp, _ = loss_and_grad(X, y, *unflatten(params - ei + ej))
            Lmm, _ = loss_and_grad(X, y, *unflatten(params - ei - ej))
            H[i, j] = (Lpp - Lpm - Lmp + Lmm) / (4 * eps**2)
    return np.linalg.eigvalsh(H)


# ---------------- 训练 ----------------
X, y = make_two_clusters()
W1, b1, W2, b2 = init_weights()
for t in range(STEPS):
    loss, (dW1, db1, dW2, db2) = loss_and_grad(X, y, W1, b1, W2, b2)
    W1 -= ETA * dW1; b1 -= ETA * db1
    W2 -= ETA * dW2; b2 -= ETA * db2
params0 = flatten((W1, b1, W2, b2))
print("训练完成: loss = %.4f" % loss)

# ---------------- 透镜一：特征空间 ----------------
h = hidden_features(X, params0)
mu = h.mean(axis=0)
Z = h - mu
cov = Z.T @ Z / len(Z)
w_, V = np.linalg.eigh(cov)
proj = Z @ V[:, ::-1][:, :2]                      # 前两个主成分投影
print("\n【透镜一】隐藏层特征 (4 维) 的前 2 主成分投影:")
print("  A 类中心 = %s" % np.round(proj[:N_CLASS].mean(axis=0), 3))
print("  B 类中心 = %s" % np.round(proj[N_CLASS:].mean(axis=0), 3))
print("  类间距离 = %.3f  (特征空间分离度)" % np.linalg.norm(
    proj[:N_CLASS].mean(axis=0) - proj[N_CLASS:].mean(axis=0)))

# ---------------- 透镜二/三：损失景观切片 ----------------
rng2 = np.random.default_rng(1)
d1 = rng2.standard_normal(params0.size); d1 /= np.linalg.norm(d1)
d2 = rng2.standard_normal(params0.size)
d2 -= d2 @ d1 * d1; d2 /= np.linalg.norm(d2)       # 与 d1 正交
xs, Lslice = loss_slice(params0, X, y, d1, d2)
print("\n【透镜二/三】损失景观切片 (过收敛点的随机 2D 平面):")
print("  切片内最小损失 = %.4f, 最大损失 = %.4f" % (Lslice.min(), Lslice.max()))
print("  (图像将显示：收敛点是否在平坦盆地中)")

# ---------------- 透镜四：收敛点曲率 ----------------
lams = hessian_at(params0, X, y)
print("\n【透镜四】收敛点 Hessian 特征值 (曲率谱):")
print("  λ_max = %.3f, λ_min = %.3f, 条件数 = %.2f" %
      (lams.max(), lams.min(), lams.max() / max(lams.min(), 1e-12)))
print("  全正特征值? %s -> 判定为%s" % (
    lams.min() > 0, "凸谷(平坦)" if lams.max() < 1 else "有尖锐方向"))

# ---------------- 可视化 ----------------
fig = plt.figure(figsize=(12, 4))
ax = fig.add_subplot(1, 3, 1)
ax.scatter(proj[:N_CLASS, 0], proj[:N_CLASS, 1], c="orange", label="A")
ax.scatter(proj[N_CLASS:, 0], proj[N_CLASS:, 1], c="blue", label="B")
ax.set_title("透镜一：特征空间 (隐藏层 PCA)"); ax.legend()

ax = fig.add_subplot(1, 3, 2)
ax.contourf(xs, xs, Lslice, levels=30, cmap="viridis")
ax.set_title("透镜二/三：损失景观切片")
ax.set_xlabel("方向 d1"); ax.set_ylabel("方向 d2")

ax = fig.add_subplot(1, 3, 3)
ax.bar(range(len(lams)), np.sort(lams)[::-1], color="steelblue")
ax.axhline(0, color="k", lw=1)
ax.set_title("透镜四：Hessian 特征值谱")
ax.set_xlabel("特征值序号"); ax.set_ylabel("λ")
plt.tight_layout()
plt.show()
```

## 层二：分步逐函数拆解

### 步骤 1：数据与模型 —— `make_two_clusters`、`init_weights`

```python

    return X, y

def init_weights(seed=SEED):
    W1 = rng.standard_normal((H, 2)) * 0.5
    ...
```

#### `make_two_clusters` / `init_weights` —— 两块可分的云与一个随机起点

**操作**：`make_two_clusters` 生成两类各 60 个 2D 点，A 围绕 $(1.5,1.5)$、B 围绕 $(-1.5,-1.5)$，各带 $\sigma = 0.8$ 的高斯散布；标签 A=1、B=0。`init_weights` 用标准正态 × 0.5 初始化 MLP 权重（$W_1: (4,2), b_1: (4,), W_2: (1,4), b_2: (1,)$）。

**对应数学**：数据"住在"两个低维簇上（§8.2.1 的流形直觉：类别结构是数据流形的连通分支）；MLP 是 $f_\theta$，学习把 $x \in \mathbb{R}^2$ 映射成概率 $\hat y$。隐藏层激活就是"中间特征空间"——透镜一分析的对象。

**输入/输出**：`make_two_clusters` 返回 `(X, y)`，`X` 形状 `(120, 2)`、`y` 形状 `(120,)`。`init_weights` 返回四个权重数组。

**可调参数**：
- `N_CLASS`（每类样本数）：改 20 数据稀疏、类边界模糊；改 200 特征更稳定。流形/聚类的"密度"由此控制。
- `H`（隐藏层神经元数）：这是"特征空间维度"。`H=2` 时特征空间是 2D，直接可视化；`H=4` 是 4D（代码用 PCA 降到 2）。`H` 越大，网络表达能力越强，但过拟合风险越高——这是"流形维度 vs 容量"的工程权衡。

### 步骤 2：前向与反向 —— `forward`、`loss_and_grad`

```python
def forward(X, W1, b1, W2, b2):
    z1 = X @ W1.T + b1
    h = np.maximum(z1, 0.0)
    z2 = h @ W2.T + b2
    yhat = 1.0 / (1.0 + np.exp(-z2))
    return z1, h, z2, yhat

def loss_and_grad(X, y, W1, b1, W2, b2):
    n = X.shape[0]
    z1, h, z2, yhat = forward(X, W1, b1, W2, b2)
    loss = -np.mean(y * np.log(yhat + 1e-12) + (1 - y) * np.log(1 - yhat + 1e-12))
    dz2 = (yhat - y.reshape(-1, 1)) / n
    dW2 = dz2.T @ h
    db2 = dz2.sum(axis=0, keepdims=True)
    dz1 = (dz2 @ W2) * (z1 > 0)
    dW1 = dz1.T @ X
    db1 = dz1.sum(axis=0)
    return loss, (dW1, db1, dW2, db2)
```

#### `forward` / `loss_and_grad` —— 前向传播与手写反向传播

**操作**：`forward` 计算 $z_1 = X W_1^\top + b_1$ → $h = \mathrm{ReLU}(z_1)$ → $z_2 = h W_2^\top + b_2$ → $\hat y = \sigma(z_2)$。`loss_and_grad` 先算二分类交叉熵，再用链式法则倒推每个权重梯度：$dW_2 = \frac{\partial L}{\partial z_2} \cdot h$、$dz_1 = (dz_2 W_2) \odot [z_1 > 0]$（ReLU 掩码）等。

**对应数学**：这就是 8.2.2 说的"梯度 = 切空间向量"的完整实现——每个参数点，反向传播给出损失函数在这一点的梯度（切空间里的最陡上升方向）。`loss_and_grad` 也被 `hessian_at` 反复调用（每次求 Hessian 需要很多次损失求值）——所以把它写成"一次算损失 + 梯度"的形式，是数值二阶方法的标准做法。

**输入**：`X (120,2)`、`y (120,)` 与四个权重数组。**输出**：`loss`（标量）与 `(dW1, db1, dW2, db2)` 四个梯度数组。

**可调参数**：这里的架构是固定的。要改，得改 `forward` 的层数/激活——比如把 ReLU 换成 `tanh`，特征空间的几何（透镜一的聚类形状）会变，但分析流程不变。这正说明：**几何体检与具体架构无关，它是对任意网络的通用视角。**

### 步骤 3：参数展平/还原 —— `flatten`、`unflatten`

```python
def flatten(params):
    W1, b1, W2, b2 = params
    return np.concatenate([W1.ravel(), b1, W2.ravel(), b2])

def unflatten(vec):
    i = 0
    W1 = vec[i:i+H*2].reshape(H, 2); i += H*2
    b1 = vec[i:i+H]; i += H
    W2 = vec[i:i+H].reshape(1, H); i += H
    b2 = vec[i:i+1]
    return W1, b1, W2, b2
```

#### `flatten` / `unflatten` —— 在"权重向量"与"权重张量"之间往返

**操作**：`flatten` 把 `(W1, b1, W2, b2)` 拍平成总长 `17` 的一维向量（$4\times2 + 4 + 4 + 1 = 17$）；`unflatten` 按同样的切片顺序还原。`i` 指针记录当前读取位置。

**对应数学**：几何分析需要一个"参数空间坐标"。`flatten` 就是把神经网络的所有参数映射成 $\mathbb{R}^{17}$ 里的一个点 $\theta$——损失函数 $L(\theta)$ 才能作为 $\mathbb{R}^{17} \to \mathbb{R}$ 的函数被分析。Hessian、随机方向切片、曲率全都定义在这个 17 维参数空间上。

**输入**：`flatten` 收四个数组，输出 `(17,)` 向量；`unflatten` 反之。**输出**：相应结构。

**可调参数**：切片边界 `i` 由 `H` 与架构硬编码。改 `H` 时这里必须同步改（这是手写代码的常见 bug 源——参数计数不一致）。更健壮的做法是用 `itertools` 或 `torch.nn.utils.parameters_to_vector` 自动展平。

### 步骤 4：损失景观切片 —— `loss_slice(params0, X, y, d1, d2, grid)`

```python
def loss_slice(params0, X, y, d1, d2, grid=40):
    xs = np.linspace(-1.5, 1.5, grid)
    L = np.zeros((grid, grid))
    for i, a in enumerate(xs):
        for j, b in enumerate(xs):
            p = unflatten(params0 + a * d1 + b * d2)
            L[i, j], _ = loss_and_grad(X, y, *p)
    return xs, L
```

#### `loss_slice` —— 在 17 维参数空间里切一片 2D 地形

**操作**：取收敛点 `params0`，沿两个正交随机方向 `d1, d2` 建立一个 2D 平面，平面上每个点对应参数 $\theta = \theta_0 + a d_1 + b d_2$，算该点的损失，填进 `grid × grid` 网格。

**对应数学**：这就是"损失景观可视化"的技术本质——高维函数没法直接画，就在某点附近取低维切片看局部形状。李浩等人 2018 年那篇《可视化损失景观》做的就是这件事（只是投影方向经过精心挑选以保持视觉真实）。切片告诉你：收敛点周围是**盆地**（四周损失升高）还是**马鞍**（一边高一边低）。

**输入**：`params0 (17,)`、数据、两个正交单位方向 `d1, d2 (17,)`、网格大小 `grid`。**输出**：`xs (grid,)`、`L (grid, grid)`。

**可调参数**：
- `d1, d2`：方向选择决定你看到的"地形"。随机方向是默认；更讲究的做法是选**主曲率方向**（Hessian 特征向量）或"数据流形方向"，能暴露最尖锐/最平坦的切面。这就是"用度量/曲率指导可视化方向"。
- `grid`：40 → 1600 次损失求值（很快）；100 → 10000 次（稍慢但更平滑）。
- `np.linspace(-1.5, 1.5, grid)` 的范围：改 ±0.5 只看到盆地局部；改 ±5 可能看到远处地形起伏。

### 步骤 5：Hessian 特征值 —— `hessian_at(params, X, y, eps)`

```python
def hessian_at(params, X, y, eps=1e-5):
    n_params = params.size
    H = np.zeros((n_params, n_params))
    for i in range(n_params):
        for j in range(n_params):
            if j < i:
                H[i, j] = H[j, i]; continue
            ei, ej = np.zeros(n_params), np.zeros(n_params)
            ei[i], ej[j] = eps, eps
            Lpp, _ = loss_and_grad(X, y, *unflatten(params + ei + ej))
            ...
            H[i, j] = (Lpp - Lpm - Lmp + Lmm) / (4 * eps**2)
    return np.linalg.eigvalsh(H)
```

#### `hessian_at` —— 数值二阶导 + 特征分解

**操作**：对 $17\times17$ 的每个元素 $H_{ij}$，用四点中心差分 $\frac{L(+\epsilon_i+\epsilon_j) - L(+\epsilon_i-\epsilon_j) - L(-\epsilon_i+\epsilon_j) + L(-\epsilon_i-\epsilon_j)}{4\epsilon^2}$ 近似二阶偏导。利用对称性 `H[i,j] = H[j,i]` 省一半计算。最后 `np.linalg.eigvalsh` 对对称矩阵求特征值。

**对应数学**：$H$ 就是 §8.2.4 的 Hessian $H(\theta) = \partial^2 L / \partial \theta^2$。它的特征值是各方向的曲率：全正 = 凸谷；有负有正 = 鞍点。输出里 `λ_min = 0.041 > 0` 判定收敛点是凸谷——这是一个**几何证明**，不是猜测。`λ_max = 2.645` 对应最尖方向的曲率，条件数 64 提示该方向优化较慢。

**输入**：`params (17,)`、数据、差分步长 `eps`。**输出**：`numpy.ndarray (17,)` 升序特征值。

**可调参数**：
- `eps`：与 8.2.4 同理——太小遇浮点噪声，太大差分误差。`1e-5` 是数值二阶导的常用甜区。
- 这个暴力双重循环对 17 参数是 289 次四点差分 ≈ 千次损失求值（瞬间完成），但参数多到百万时完全不可行——这正是真实深度学习里要改用**随机特征分解**（只求最大/最小特征值）的原因。`hessian_eigenthings` 库就是干这个的。

LAM = 0.1
